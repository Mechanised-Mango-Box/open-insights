# What the API server needs at runtime, and nothing else. Model training pulls a
# heavier set - see requirements-training.txt, which nothing under the request
# path imports.
#
# Pinned rather than floating because there is no lockfile: the container is
# rebuilt from this file on a host whose wheels must match the ones this was
# tested against. Direct dependencies only - pip resolves the rest.

# Transcription. Deliberately NOT openai-whisper: CTranslate2 reaches a cloud GPU
# by switching WHISPER_DEVICE/WHISPER_COMPUTE_TYPE rather than a rewrite, and it
# keeps torch (plus ~19 nvidia-* CUDA wheels, 3-4GB) out of the container image.
faster-whisper==1.2.1

flask==3.1.3
flask-cors==6.0.5

# Per-IP rate limiting for the public API key tier. Its in-memory storage is the
# right backend here precisely because the server is single-process by design
# (see the -w 1 note in the Dockerfile) - a shared store would be solving a
# problem this deployment does not have.
flask-limiter==4.1.1

# WSGI server. The Werkzeug one main.py uses is fine for a desktop and is not
# meant to face the internet.
gunicorn==23.0.0

# On-screen text (text_stats.py). rapidocr itself is NOT here: it declares a
# dependency on opencv-python, which would install beside the headless build
# below, overwrite its cv2, and need libGL - which the slim image does not have.
# It goes in with --no-deps from requirements-nodeps.txt, and these are the rest
# of what it needs. onnxruntime, its inference engine, already comes with
# faster-whisper, as do PyYAML, requests and tqdm.
colorlog==6.12.0
omegaconf==2.3.1
pillow==12.3.0
pyclipper==1.4.0
shapely==2.1.2
six==1.17.0

# Audio stats (audio_stats.py): Praat's pitch tracker, for pitch variation. Its
# only dependency is numpy. Silero VAD, the other half, ships with faster-whisper.
praat-parselmouth==0.4.7

# Scene stats. headless rather than opencv-python: the server never opens a
# window, and it drops the GUI shared-library dependencies from the image.
opencv-python-headless==5.0.0.93

# Checking an upload is a video, and decoding what OpenCV cannot (video_files.py).
# faster-whisper brings it anyway; pinned here because the server now imports it
# directly and relies on the software AV1 decoder (dav1d) its wheels carry - the
# opencv-python wheels decode AV1 only through hardware.
av==18.1.0

# The engagement model behind /api/videos/<hash>/recommendation (inference.py).
# Runtime rather than training-only: the server loads the pickled random forest
# and regression at startup. That bundle is committed (engagement_model/) and was
# made under these exact pins - a scikit-learn pickle does not load reliably under
# another version, so changing any of them means rerunning
# scripts/train_engagement_model.py <export> and committing the new bundle.
joblib==1.5.3
numpy==2.5.1
pandas==3.0.5
scikit-learn==1.9.0
